# Step 1: Import libraries
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Step 2: Load dataset
df = pd.read_csv("sales_data_sample.csv")

print(df.head())
print(df.shape)
print(df.columns)

# Step 3: Select features and remove missing values
X = df[["SALES", "QUANTITYORDERED", "PRICEEACH", "MSRP"]]
X = X.dropna()

print(X.head())

# Step 4: Standardize the data
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 5: Elbow Method
wcss = []

for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(X_scaled)
    wcss.append(model.inertia_)

plt.plot(range(1, 11), wcss, marker="o")
plt.title("Elbow Method")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS")
plt.xticks(range(1, 11))
plt.grid(True)
plt.show()

# Step 6: Apply K-Means
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
labels = kmeans.fit_predict(X_scaled)

# Assign cluster labels to the matching original rows
df["Cluster"] = pd.Series(labels, index=X.index)

print(df[["SALES", "QUANTITYORDERED", "Cluster"]].head(10))

# Step 7: Visualize the clusters
plt.scatter(
    X["SALES"],
    X["QUANTITYORDERED"],
    c=labels
)

plt.title("K-Means Clustering of Sales Data")
plt.xlabel("Sales")
plt.ylabel("Quantity Ordered")
plt.show()

# Step 8: Display records in each cluster
print("Number of records in each cluster:")
print(df["Cluster"].value_counts().sort_index())
